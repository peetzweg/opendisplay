cmake_minimum_required(VERSION 3.16)
project(iphone_native_mirror C OBJC)
set(CMAKE_C_STANDARD 11)
option(MIRROR_INTERNAL_MDNS "Use independent mDNS so macOS can discover this local receiver" ON)
if(MIRROR_INTERNAL_MDNS)
  add_subdirectory(vendor/uxplay/lib/mdnsd)
else()
  add_compile_definitions(EXTERNAL_DNS_SD UXPLAY_HAVE_APPLE_P2P)
  add_subdirectory(vendor/uxplay/lib/dns_sd)
endif()
add_subdirectory(vendor/uxplay/lib/llhttp)
add_subdirectory(vendor/uxplay/lib/playfair)
add_subdirectory(vendor/uxplay/lib)
option(MIRROR_MODERN_PROFILE "Use the modern AirPlay compatibility profile" ON)
set(MIRROR_MODEL "AppleTV6,2" CACHE STRING "AirPlay receiver discovery model; MacBook identity fails native FairPlay negotiation")
set(MIRROR_PROTOCOL_MODEL "AppleTV6,2" CACHE STRING "Compatibility model returned by the AirPlay protocol")
if(MIRROR_MODERN_PROFILE)
  target_compile_definitions(airplay PRIVATE GLOBAL_MODEL="${MIRROR_PROTOCOL_MODEL}" GLOBAL_VERSION="380.20.1")
  target_compile_definitions(dnssd PRIVATE GLOBAL_MODEL="${MIRROR_MODEL}" GLOBAL_VERSION="380.20.1")
endif()
add_executable(airplay-bridge native/receiver.c native/audio-route.c native/audio-route-policy.c)
target_include_directories(airplay-bridge PRIVATE vendor/uxplay/lib)
target_link_libraries(airplay-bridge airplay "-framework CoreAudio" "-framework CoreFoundation")
add_executable(iphone-display-error native/show-error.m)
target_compile_options(iphone-display-error PRIVATE -fobjc-arc)
target_link_libraries(iphone-display-error "-framework AppKit")
option(MIRROR_BUILD_TESTS "Build native regression checks" OFF)
if(MIRROR_BUILD_TESTS)
  enable_testing()
  add_executable(empty-config-test native/test-empty-config.c)
  target_include_directories(empty-config-test PRIVATE vendor/uxplay/lib)
  target_link_libraries(empty-config-test airplay)
  add_test(NAME empty-config COMMAND empty-config-test)
  set_tests_properties(empty-config PROPERTIES TIMEOUT 10)
  add_executable(feature-bits-test native/test-feature-bits.c)
  target_include_directories(feature-bits-test PRIVATE vendor/uxplay/lib)
  target_link_libraries(feature-bits-test airplay "-framework CoreAudio" "-framework CoreFoundation")
  add_test(NAME feature-bits COMMAND feature-bits-test)
  set_tests_properties(feature-bits PROPERTIES TIMEOUT 10)
endif()
